/**
 * WebAR - JSARToolKit5 + Three.js
 * 真正的 Hiro Marker 检测 + FBX 模型叠加
 */
import { useEffect, useRef, useState } from 'react'
import { View, Text } from '@tarojs/components'
import Taro from '@tarojs/taro'
import './viewer.scss'

const MODEL_URL = '/blackhole-3d-xr/models/test.fbx'
const MARKER_PATTERN = '/blackhole-3d-xr/artoolkit/patt.hiro'
const CAMERA_PARAM = '/blackhole-3d-xr/artoolkit/camera_para-iPhone%205%20rear%20640x480%201.0m.dat'

type Status = 'idle' | 'loading' | 'inited' | 'model-loading' | 'ready' | 'error'

export default function Viewer() {
  const [status, setStatus] = useState<Status>('idle')
  const [msg, setMsg] = useState('点击「启动 AR」开始扫描')

  // 页面卸载清理
  useEffect(() => {
    return () => {
      // 停止所有 MediaStream tracks
      document.querySelectorAll('video').forEach(v => {
        (v.srcObject as MediaStream)?.getTracks().forEach(t => t.stop())
      })
    }
  }, [])

  const startAR = async () => {
    try {
      setStatus('loading')
      setMsg('初始化 AR 引擎...')

      // ═══ 1. 等待 ARController 全局变量可用 ════
      await waitForARController()

      // ═══ 2. 初始化 AR 场景 ════
      const arScene = await initARScene()

      setMsg('AR 就绪，加载 3D 模型...')
      setStatus('model-loading')

      // ═══ 3. 加载 Three.js + FBX ═══
      const THREE = await import('three')
      const { FBXLoader } = await import('three/examples/jsm/loaders/FBXLoader.js')

      // ═══ 4. 加载 FBX 模型 ═══
      let fbxModel: any = null

      try {
        fbxModel = await loadFBX(THREE, FBXLoader, MODEL_URL)
      } catch (err) {
        console.warn('[AR] FBX load failed, using fallback cube:', err)
        // FBX 加载失败时用立方体代替
        fbxModel = createFallbackCube(THREE)
      }

      // ═══ 5. 加载 Hiro Marker 并绑定模型 ═══
      await loadMarkerAndBindModel(arScene, fbxModel)

      setStatus('ready')
      setMsg('✅ 检测到识别图！')

    } catch (err: any) {
      console.error('[AR] Init error:', err)
      setStatus('error')
      setMsg('AR 初始化失败: ' + (err?.message || String(err)))
      Taro.showToast({ title: 'AR 启动失败', icon: 'none', duration: 2000 })
    }
  }

  // ─── 等待 ARController 全局变量 ───────────────────────────────────────
  const waitForARController = (timeout = 10000): Promise<void> => {
    return new Promise((resolve, reject) => {
      if ((window as any).ARController) {
        resolve()
        return
      }
      const timer = setTimeout(() => {
        window.removeEventListener('ARController-loaded', onLoad)
        reject(new Error('ARController 加载超时，请检查网络连接'))
      }, timeout)
      const onLoad = () => {
        clearTimeout(timer)
        resolve()
      }
      window.addEventListener('ARController-loaded', onLoad)
      // 如果已经加载完成
      if ((window as any).ARController) {
        clearTimeout(timer)
        resolve()
      }
    })
  }

  // ─── 初始化 AR 场景 ─────────────────────────────────────────────────
  const initARScene = (): Promise<any> => {
    return new Promise((resolve, reject) => {
      const ARController = (window as any).ARController
      if (!ARController) {
        reject(new Error('ARController 未找到'))
        return
      }

      ARController.getUserMediaThreeScene({
        maxARVideoSize: 640,
        cameraParam: CAMERA_PARAM,
        onSuccess: (arScene: any, arController: any, arCamera: any) => {
          console.log('[AR] Scene inited, video:', arController.videoWidth, 'x', arController.videoHeight)

          document.body.className = arController.orientation || ''

          const renderer = new (window as any).THREE.WebGLRenderer({ antialias: true })
          const isPortrait = arController.orientation === 'portrait'

          if (isPortrait) {
            const w = (window.innerWidth / arController.videoHeight) * arController.videoWidth
            const h = window.innerWidth
            renderer.setSize(w, h)
            renderer.domElement.style.paddingBottom = (w - h) + 'px'
          } else {
            const isMobile = /Android|mobile|iPad|iPhone/i.test(navigator.userAgent)
            if (isMobile) {
              renderer.setSize(window.innerWidth, (window.innerWidth / arController.videoWidth) * arController.videoHeight)
            } else {
              renderer.setSize(arController.videoWidth, arController.videoHeight)
              document.body.className += ' desktop'
            }
          }

          // canvas 放在 #ar-container 内部
          const container = document.getElementById('ar-container')
          if (container) {
            container.appendChild(renderer.domElement)
          } else {
            document.body.insertBefore(renderer.domElement, document.body.firstChild)
          }

          // 渲染循环
          const tick = () => {
            arScene.process()
            arScene.renderOn(renderer)
            requestAnimationFrame(tick)
          }
          tick()

          // 保存引用用于清理
          ;(window as any).__arRenderer = renderer
          ;(window as any).__arController = arController

          setStatus('inited')
          resolve(arScene)
        },
        onError: (err: any) => {
          console.error('[AR] getUserMedia error:', err)
          reject(new Error('摄像头初始化失败'))
        }
      })
    })
  }

  // ─── 加载 FBX 模型 ──────────────────────────────────────────────────
  const loadFBX = (THREE: any, FBXLoader: any, url: string): Promise<any> => {
    return new Promise((resolve, reject) => {
      const loader = new FBXLoader()
      loader.load(
        url,
        (group: any) => {
          // 自动居中 + 缩放
          const box = new THREE.Box3().setFromObject(group)
          const size = box.getSize(new THREE.Vector3())
          const max = Math.max(size.x, size.y, size.z)
          group.scale.setScalar(1.0 / max)

          // 放在 marker 表面
          group.position.set(0, 0, 0.5)

          group.traverse((child: any) => {
            if (child.isMesh) {
              child.castShadow = true
            }
          })

          resolve(group)
        },
        undefined,
        (err: any) => reject(err)
      )
    })
  }

  // ─── 备用几何体（FBX 加载失败时） ────────────────────────────────────
  const createFallbackCube = (THREE: any): any => {
    const geometry = new THREE.BoxGeometry(0.5, 0.5, 0.5)
    const material = new THREE.MeshNormalMaterial()
    const cube = new THREE.Mesh(geometry, material)
    cube.position.set(0, 0, 0.5)
    return cube
  }

  // ─── 加载 Hiro Marker 并绑定模型 ─────────────────────────────────────
  const loadMarkerAndBindModel = (arScene: any, model: any): Promise<void> => {
    return new Promise((resolve, reject) => {
      const arController = (window as any).__arController

      arController.loadMarker(MARKER_PATTERN, (markerId: number) => {
        console.log('[AR] Hiro marker loaded, id:', markerId)

        // 创建 marker root（Three.js Group，会跟随 marker 移动）
        const markerRoot = arController.createThreeMarker(markerId)

        // 把模型添加到 marker root
        markerRoot.add(model)
        arScene.scene.add(markerRoot)

        Taro.showToast({ title: '识别图检测就绪', icon: 'none', duration: 1500 })
        resolve()
      })
    })
  }

  // ─── 状态显示 ───────────────────────────────────────────────────────
  const statusColor = {
    idle: '#888',
    loading: '#f5a623',
    inited: '#7ed321',
    'model-loading': '#f5a623',
    ready: '#7ed321',
    error: '#d0021b'
  }[status] || '#888'

  const statusText = {
    idle: '待启动',
    loading: '初始化中',
    inited: 'AR 已就绪',
    'model-loading': '加载模型',
    ready: '扫描识别图',
    error: '错误'
  }[status] || status

  return (
    <View className="viewer-page">
      {/* AR 渲染容器（JSARToolKit 会把 canvas 放这里） */}
      <View id="ar-container" className="ar-container" />

      {/* 顶部提示 */}
      <View className="camera-hint">
        <Text className="hint-text">📷 JSARToolKit5 Marker AR</Text>
        <Text className="hint-sub">摄像头 + Hiro 识别图 + FBX 模型</Text>
      </View>

      {/* 状态显示 */}
      {(status !== 'idle') && (
        <View className="status-overlay" style={{ background: status === 'error' ? 'rgba(208,2,27,0.85)' : 'rgba(0,0,0,0.6)' }}>
          <Text className="status-text" style={{ color: '#fff' }}>{statusText}</Text>
          <Text className="progress-text" style={{ color: '#fff', opacity: 0.8 }}>{msg}</Text>
        </View>
      )}

      {/* 控制栏 */}
      <View className="control-bar">
        <View className="status-row">
          <View className="status-dot" style={{ background: statusColor }} />
          <Text className="status-label">{statusText}</Text>
        </View>
        <View className="btn-group">
          {status === 'idle' && (
            <View className="btn primary" onClick={startAR}>启动 AR</View>
          )}
          {status === 'error' && (
            <View className="btn warn" onClick={startAR}>重新启动</View>
          )}
        </View>
      </View>

      {/* Hiro 识别图提示 */}
      {status === 'ready' && (
        <View className="marker-guide">
          <View className="marker-frame" />
          <Text className="guide-text">将 Hiro 识别图对准摄像头</Text>
          <Text className="guide-hint">
            👉 {MARKER_PATTERN.replace('/blackhole-3d-xr', '')}
          </Text>
        </View>
      )}
    </View>
  )
}
